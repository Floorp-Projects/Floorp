cmake_minimum_required(VERSION 3.20)
project(FloorpPrivateProfileToken LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The Windows native token contract must run on Windows")
endif()
add_library(floorp_private_profile_token STATIC PrivateProfileToken.cpp)
target_compile_features(floorp_private_profile_token PUBLIC cxx_std_17)
# Exercise the same feature-macro branch as the real Gecko translation unit.
target_compile_definitions(floorp_private_profile_token PRIVATE MOZILLA_INTERNAL_API UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(floorp_private_profile_token PUBLIC advapi32)
target_include_directories(floorp_private_profile_token PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
  target_compile_options(floorp_private_profile_token PRIVATE /W4 /WX /permissive-)
endif()
enable_testing()
add_executable(floorp_private_profile_token_tests Tests.cpp)
target_link_libraries(floorp_private_profile_token_tests PRIVATE floorp_private_profile_token)
target_compile_definitions(floorp_private_profile_token_tests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME windows_profile_token_native COMMAND floorp_private_profile_token_tests)
set_tests_properties(windows_profile_token_native PROPERTIES TIMEOUT 60)
